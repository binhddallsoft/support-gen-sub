# numpy 1.26.4 là bản 1.x cuối cùng, có sẵn gói biên dịch cho Python 3.12 trên
# Intel Mac. Giữ ở nhánh 1.x là lựa chọn an toàn hai chiều: thư viện biên dịch
# theo numpy 1 thì bắt buộc phải có numpy 1, còn thư viện biên dịch theo
# numpy 2 vẫn chạy được với numpy 1.19 trở lên.
numpy==1.26.4

# CTranslate2 là bộ máy tính toán của faster-whisper (chạy CPU, kiểu int8).
# 4.3.1 là bản CUỐI CÙNG của nhánh 4.0-4.5 còn phát hành gói
# macosx_10_9_x86_64 cho Python 3.12. Bản 4.4.0 và 4.5.0 chỉ có gói cho Apple
# Silicon (macosx_11_0_arm64) và KHÔNG có bản mã nguồn, nên trên máy Intel cả
# lệnh cài sẽ đứt ngay ở dòng này. Đã kiểm lại trên PyPI: các bản có gói cho
# Intel Mac + Python 3.12 là 3.19.0-4.3.1, rồi 4.6.0 trở lên.
# Muốn dùng bản mới hơn thì phải nhảy hẳn lên 4.6.0 (macosx_10_13_x86_64).
ctranslate2==4.3.1

# Dùng cho bộ lọc VAD (bỏ qua đoạn im lặng, chống Whisper "nghe ra chữ" ở đoạn
# chỉ có nhạc). 1.19.2 là bản còn phát hành cho macOS Intel.
onnxruntime==1.19.2

# Giải mã âm thanh trong faster-whisper. Có ffmpeg bọc sẵn bên trong, tách biệt
# với file ffmpeg mà CaiDat.command tải về thư mục app.
av==12.3.0

# Bộ tách token của mô hình Whisper (viết bằng Rust, có gói biên dịch sẵn).
tokenizers==0.20.3

# Tải mô hình về máy. Bản này hỗ trợ tải nối tiếp khi đứt mạng giữa chừng —
# đó là lý do CaiDat.command chạy lại lần hai không phải tải lại từ đầu.
huggingface-hub==0.26.5
tqdm==4.67.1

# Bộ nhận dạng giọng nói. Trên máy này luôn chạy CPU với compute_type=int8.
# 1.1.x là bản đầu tiên nhận tên mô hình "large-v3-turbo" — mô hình mặc định

faster-whisper==1.1.1
